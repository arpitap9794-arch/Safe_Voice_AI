# STEP 2: AI COMMAND DETECTION
def analyze_command(text):
    text = text.lower().strip()

    emergency_phrases = [
        "help me", "i need help", "emergency",
        "save me", "sos", "i am in danger",
        "bachao", "madad karo", "mujhe help chahiye"
    ]

    if any(phrase in text for phrase in emergency_phrases):
        return "SOS"

    return "NORMAL"


# STEP 3: VOICE RECOGNITION
def listen_to_voice():
    recognizer = sr.Recognizer()

    try:
        with sr.Microphone() as source:
            log("Listening... Speak now.")
            recognizer.adjust_for_ambient_noise(source, duration=1)
            audio = recognizer.listen(
                source, timeout=5, phrase_time_limit=8
            )

        return recognizer.recognize_google(audio)

    except Exception as error:
        return "ERROR: " + str(error)

# STEP 7: BATTERY PROTECTION
def get_battery_status():
    try:
        battery = psutil.sensors_battery()

        if battery is None:
            return {
                "percentage": "Unavailable",
                "charging": "Unknown",
                "status": "Battery information unavailable"
            }

        percent = battery.percent
        charging = battery.power_plugged

        if percent <= 10 and not charging:
            status = "Critical"
        elif percent <= 20 and not charging:
            status = "Low"
        else:
            status = "Normal"

        return {
            "percentage": percent,
            "charging": charging,
            "status": status
        }

    except Exception:
        return {
            "percentage": "Unavailable",
            "charging": "Unknown",
            "status": "Battery information unavailable"
        }

        print("AI result:", command)

            trigger_sos("Voice command detected: " + result)
        else:
            print("Normal command detected.")
